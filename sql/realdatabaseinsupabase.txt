**product_performance Table**
create view public.product_performance as
select
  p.id,
  p.name,
  p.category,
  count(s.id) as times_sold,
  COALESCE(
    sum((item.value ->> 'quantity'::text)::integer),
    0::bigint
  ) as total_quantity_sold,
  COALESCE(
    sum(
      ((item.value ->> 'quantity'::text)::numeric) * ((item.value ->> 'price'::text)::numeric)
    ),
    0::numeric
  ) as total_revenue
from
  products p
  left join sales s on true
  left join lateral jsonb_array_elements(s.items) item (value) on ((item.value ->> 'product_id'::text)::uuid) = p.id
group by
  p.id,
  p.name,
  p.category;

**expense_summaries Table**
create view public.expense_summary as
select
  category,
  count(*) as transaction_count,
  sum(amount) as total_amount,
  avg(amount) as average_amount
from
  expenses
group by
  category
order by
  (sum(amount)) desc;

**daily_sales_summary Table**
create view public.daily_sales_summary as
select
  date (created_at) as sale_date,
  count(*) as total_transactions,
  sum(total) as total_revenue,
  avg(total) as average_sale
from
  sales
group by
  (date (created_at))
order by
  (date (created_at)) desc;
